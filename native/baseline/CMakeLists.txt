cmake_minimum_required(VERSION 3.20)
project(TSRBaseline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(tsr_reference_tests reference_tests.cpp)
add_test(NAME reference COMMAND tsr_reference_tests)
if(WIN32)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Bilinear.hlsl" BILINEAR_SOURCE)
  configure_file(shader_source.h.in shader_source.h @ONLY)
  add_executable(tsr_dx12_baseline dx12_baseline.cpp)
  target_include_directories(tsr_dx12_baseline PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(tsr_dx12_baseline PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(tsr_dx12_baseline PRIVATE d3d12 dxgi d3dcompiler)
  if(MSVC)
    target_compile_options(tsr_dx12_baseline PRIVATE /W4 /permissive-)
  endif()
  add_test(NAME dx12_warp COMMAND tsr_dx12_baseline --warp)
  set_tests_properties(dx12_warp PROPERTIES TIMEOUT 120)
endif()
